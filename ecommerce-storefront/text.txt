// themes/marketplace/Home.tsx
import { TenantConfig } from '@/themes/types'
import { getProductsByTenant } from '@/lib/api/products'
import { Header } from './Header'
import { Footer } from './Footer'
import { ProductCard } from './ProductCard'

export async function Home({ tenant }: { tenant: TenantConfig }) {
  const products = await getProductsByTenant(tenant.tenantId)

  return (
    <div>
      <Header tenant={tenant} />
      <main style={{ padding: 16, display: 'grid', gap: 12, gridTemplateColumns: 'repeat(auto-fill, minmax(160px, 1fr))' }}>
        {products.map(p => (
          <ProductCard key={p.id} name={p.name} price={p.price} />
        ))}
      </main>
      <Footer />
    </div>
  )
}

// middleware.ts:
import { NextRequest, NextResponse } from 'next/server'
export function middleware(req: NextRequest) {
  const host = (req.headers.get('host') || '').split(':')[0]
  const res = NextResponse.next()
  res.headers.set('x-tenant-host', host)
  return res
}

// app/page.tsx:
import { headers } from 'next/headers'
import { getTenantByHost } from '@/lib/api/tenant'
import { themeRegistry } from '@/themes/registry'
import type { Metadata } from 'next'

export async function generateMetadata(): Promise<Metadata> {
  const headersList = await headers()
  const host = headersList.get('x-tenant-host') || 'shop1.localhost'
  const tenant = await getTenantByHost(host)

  return {
    title: tenant.storeName,
    description: `Shop at ${tenant.storeName}`,
  }
}

export default async function HomePage() {
  const headersList = await headers()
  const host = headersList.get('x-tenant-host') || 'shop1.localhost'
  const tenant = await getTenantByHost(host)
  const ThemeModule = await themeRegistry[tenant.theme]()
  const Home = ThemeModule.Home
  return <Home tenant={tenant} />
}
